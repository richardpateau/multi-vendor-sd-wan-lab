            set server 10.0.1.1
            set members 1 2
            set interval 200
            set failure-threshold 3
            set recovery-threshold 3
        next
    end
    
    config service
        edit 1
            set name "HQ-to-Branch-SLA"
            set dst "Branch-Subnet"
            set src "all"
            set mode sla
            set sla "ISP-Link-Monitor"
            config members
                edit 1
                    set interface "port2"
                    set priority 1
                next
                edit 2
                    set interface "port3"
                    set priority 2
                next
            end
        next
    end
end

config system sdwan
    set status enable

    config health-check
        edit ISP_A_Monitor
            set server "10.0.1.1"
            set protocol ping
            set members 1
            set interval 5000
            set failtime 3
            set recoverytime 3
        next

        edit ISP_B_Monitor
            set server "10.0.2.1"
            set protocol ping
            set members 2
            set interval 500
            set failtime 3
            set recoverytime 3
        next
    end
end

config system sdwan
    config service
        edit 1
            set name "HQ-to-Branch"
            set mode manual
            set dst "Branch-Subnet"
            set src "all"
            set mode priority
            set health-check ISP_A_Monitor ISP_B_Monitor
            set priority-members 1 2
            set status enable
        next
    end
end
